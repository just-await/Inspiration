// supabase/functions/telegram-bot/index.ts
import { serve } from "https://deno.land/std@0.168.0/http/server.ts"
import { createClient } from 'https://esm.sh/@supabase/supabase-js@2'

const BOT_TOKEN = Deno.env.get('TELEGRAM_BOT_TOKEN')!
const ADMIN_CHAT_ID = Deno.env.get('ADMIN_CHAT_ID')!
const SUPABASE_URL = Deno.env.get('SUPABASE_URL')!
const SERVICE_ROLE_KEY = Deno.env.get('MY_SERVICE_KEY') ?? Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!

const supabaseAdmin = createClient(SUPABASE_URL, SERVICE_ROLE_KEY)

const CATEGORIES: Record<string, string> = {
    'motivation': '🔥 Мотивация',
    'humor': '😂 Юмор',
    'wisdom': '🦉 Мудрость',
    'absurd': '🤪 Абсурд',
    'love': '❤️ Любовь',
    'media': '🎬 Из медиа',
    'thoughts': '💭 Мысли вслух'
};

serve(async (req) => {
  try {
    const url = new URL(req.url)
    const body = await req.json().catch(() => ({}))

    // =================================================================
    // 1. ОБРАБОТКА ТЕКСТОВЫХ СООБЩЕНИЙ
    // =================================================================
    if (body.message) {
        const msg = body.message
        const chatId = msg.chat.id
        const text = msg.text

        if (text && text.startsWith('/start')) {
            const param = text.split(' ')[1]
            if (param) {
                const { data: quotesData } = await supabaseAdmin.from('quotes').update({ telegram_id: chatId }).eq('session_id', param).select()
                const count = quotesData ? quotesData.length : 0
                await sendTelegram(chatId, `👋 <b>Привет!</b>\n\nЯ связал твой аккаунт с твоими заявками (найдено: ${count}).\nЖди уведомлений!`)
            } else {
                await sendTelegram(chatId, `Привет! Предлагай фразы на сайте, а я буду присылать уведомления.`)
            }
            return new Response("OK")
        }

        if (msg.reply_to_message && msg.reply_to_message.text && msg.reply_to_message.text.includes('Заявка #')) {
            const match = msg.reply_to_message.text.match(/Заявка #(\d+)/)
            if (match && match[1]) {
                const quoteId = match[1]
                const reason = text 

                const { data: record } = await supabaseAdmin.from('quotes').update({ 
                        is_approved: false, status: 'rejected', rejection_reason: reason 
                    }).eq('id', quoteId).select('*').single()

                if (record) {
                    if (record.telegram_id) {
                        const userMsg = `😔 <b>Твоя фраза отклонена</b>\n\n<i>"${record.text}"</i>\n\n<b>Причина:</b> ${reason}\n\nВы можете отредактировать её в личном кабинете!`
                        await sendTelegram(record.telegram_id, userMsg)
                    }
                    const buttons = getAdminButtons(quoteId, record.contact)
                    await editTelegramMessage(chatId, msg.reply_to_message.message_id, formatAdminMessage(record, "❌ ОТКЛОНЕНО (Скрыто)"), buttons)
                }
            }
            return new Response("OK")
        }
    }

    // =================================================================
    // 2. ОБРАБОТКА КНОПОК
    // =================================================================
    if (body.callback_query) {
      const callback = body.callback_query
      const dataParts = callback.data.split('_')
      
      let action, quoteId
      if (dataParts.length === 2) {[action, quoteId] = dataParts
      } else {
          action = dataParts[0] + '_' + dataParts[1]
          quoteId = dataParts[2]
      }

      const messageId = callback.message.message_id
      const chatId = callback.message.chat.id

      const { data: record } = await supabaseAdmin.from('quotes').select('*').eq('id', quoteId).single()
      if (!record) return new Response("OK")

      if (action === 'changecat') {
          const categoriesKeyboard = [[{ text: "🔥 Мотивация", callback_data: `setcat_motivation_${quoteId}` }, { text: "😂 Юмор", callback_data: `setcat_humor_${quoteId}` }],[{ text: "🦉 Мудрость", callback_data: `setcat_wisdom_${quoteId}` }, { text: "🤪 Абсурд", callback_data: `setcat_absurd_${quoteId}` }],[{ text: "❤️ Любовь", callback_data: `setcat_love_${quoteId}` }, { text: "🎬 Из медиа", callback_data: `setcat_media_${quoteId}` }],[{ text: "💭 Мысли вслух", callback_data: `setcat_thoughts_${quoteId}` }],[{ text: "« Отмена", callback_data: `cancelcat_null_${quoteId}` }]
          ]
          await editTelegramMessage(chatId, messageId, formatAdminMessage(record, "🔄 <b>Выберите новую категорию:</b>"), categoriesKeyboard)
          await answerCallback(callback.id)
          return new Response("OK")
      }

      if (action === 'cancelcat_null') {
          await editTelegramMessage(chatId, messageId, formatAdminMessage(record, "⏳ Ожидает решения"), getAdminButtons(quoteId, record.contact))
          await answerCallback(callback.id)
          return new Response("OK")
      }

      if (action.startsWith('setcat_')) {
          const selectedCat = action.split('_')[1];
          const catName = CATEGORIES[selectedCat] || 'Другое';
          
          const adminNote = `Модератор заботливо перенес вашу фразу в категорию «${catName}» 😉`;

          await supabaseAdmin.from('quotes').update({
              category: selectedCat,
              is_approved: true,
              status: 'approved',
              admin_note: adminNote,
              rejection_reason: null
          }).eq('id', quoteId);

          if (record.telegram_id) {
              const userMsg = `🎉 <b>Твоя фраза одобрена и добавлена на сайт!</b>\n\n<i>"${record.text}"</i>\n\nP.S. ${adminNote}`;
              await sendTelegram(record.telegram_id, userMsg).catch(e => console.error(e));
          }

          record.category = selectedCat;
          record.is_approved = true;
          record.status = 'approved';

          await editTelegramMessage(chatId, messageId, formatAdminMessage(record, "✅ ОДОБРЕНО И ПЕРЕНЕСЕНО"), getAdminButtons(quoteId, record.contact));
          await answerCallback(callback.id, "Категория изменена");
          return new Response("OK");
      }

      if (action === 'reject') {
          const askReasonText = formatAdminMessage(record, "✍️ <b>Введи причину ОТВЕТОМ (Reply) на это сообщение</b>\nили нажми кнопку ниже")
          const skipButton = [[{ text: "🤷‍♂️ Не называть причину", callback_data: `skip_reject_${quoteId}` }]]
          await editTelegramMessage(chatId, messageId, askReasonText, skipButton)
          await answerCallback(callback.id)
          return new Response("OK")
      }

      let statusText = ""
      let isApproved = false
      let dbStatus = "pending"

      if (action === 'approve') {
        isApproved = true; dbStatus = 'approved'; statusText = "✅ ОДОБРЕНО (Видно на сайте)"
      } 
      if (action === 'skip_reject') {
        isApproved = false; dbStatus = 'rejected'; statusText = "❌ ОТКЛОНЕНО (Скрыто)"
      }

      await supabaseAdmin.from('quotes').update({ 
            is_approved: isApproved, status: dbStatus, rejection_reason: null, admin_note: null
      }).eq('id', quoteId)

      if (record.telegram_id) {
          const userMsg = isApproved 
            ? `🎉 <b>Твоя фраза одобрена!</b>\n\n<i>"${record.text}"</i>\n\nСкоро она появится в ротации на сайте.`
            : `😔 <b>Твоя фраза отклонена</b>\n\n<i>"${record.text}"</i>\n\nВы можете отредактировать её в личном кабинете!`;
          await sendTelegram(record.telegram_id, userMsg).catch(e => console.error(e))
      }

      const buttons = getAdminButtons(quoteId, record.contact)
      await editTelegramMessage(chatId, messageId, formatAdminMessage(record, statusText), buttons)
      await answerCallback(callback.id, "Статус обновлен")

      return new Response("OK")
    }

    // =================================================================
    // 3. НОВАЯ ИЛИ ОБНОВЛЕННАЯ ЗАЯВКА (WEBHOOK)
    // =================================================================
    if ((body.type === 'INSERT' || body.type === 'UPDATE') && body.table === 'quotes') {
      const record = body.record
      
      // Отправляем уведомление, только если это новая заявка, ИЛИ если заявка была отредактирована и снова стала pending
      if (body.type === 'INSERT' || (body.type === 'UPDATE' && record.status === 'pending')) {
          const title = body.type === 'UPDATE' ? "🔄 <b>Обновленная заявка</b>" : "⏳ <b>Ожидает решения</b>";
          await sendTelegram(ADMIN_CHAT_ID, formatAdminMessage(record, title), getAdminButtons(record.id, record.contact))
      }
      return new Response("OK")
    }

    return new Response("OK")
  } catch (error) {
    return new Response(JSON.stringify({ error: error.message }), { status: 500 })
  }
})

function formatAdminMessage(record: any, status: string) {
    const catName = CATEGORIES[record.category] || record.category || 'Не указана';
    const authorType = record.is_external_author ? '(Из медиа / Чужая)' : '(Личная фраза)';
    return `📝 <b>Заявка #${record.id}</b>\n` +
           `🏷 Категория: <b>${catName}</b>\n\n` +
           `<code>${record.text}</code>\n\n` +
           `Автор: <b>${record.author}</b> ${authorType}\n` +
           `Статус: ${status}`
}

function getAdminButtons(quoteId: any, contact: any) {
    const inlineKeyboard = [[
            { text: "✅ Одобрить", callback_data: `approve_${quoteId}` },
            { text: "❌ Отклонить", callback_data: `reject_${quoteId}` }
        ],[
            { text: "🔄 Изменить категорию", callback_data: `changecat_${quoteId}` }
        ]
    ]
    if (contact) inlineKeyboard.push([{ text: `✈️ ЛС @${contact}`, url: `https://t.me/${contact}` }])
    return inlineKeyboard
}

async function sendTelegram(chatId: any, text: string, buttons: any = null) {
    const body: any = { chat_id: chatId, text: text, parse_mode: 'HTML' }
    if (buttons) body.reply_markup = { inline_keyboard: buttons }
    return await fetch(`https://api.telegram.org/bot${BOT_TOKEN}/sendMessage`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body) })
}

async function editTelegramMessage(chatId: any, messageId: any, text: string, buttons: any) {
    return await fetch(`https://api.telegram.org/bot${BOT_TOKEN}/editMessageText`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ chat_id: chatId, message_id: messageId, text: text, parse_mode: 'HTML', reply_markup: { inline_keyboard: buttons } }) })
}

async function answerCallback(callbackId: any, text: string = "") {
    return await fetch(`https://api.telegram.org/bot${BOT_TOKEN}/answerCallbackQuery`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ callback_query_id: callbackId, text: text }) })
}
